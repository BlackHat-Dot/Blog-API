from pydantic import ConfigDict, Field, PostgresDsn
from pydantic_settings import BaseSettings


class Settings(BaseSettings):
    app_name: str = "Longhand API"
    api_v1_str: str = "/api/v1"
    secret_key: str = Field(..., validation_alias="SECRET_KEY")
    access_token_expire_minutes: int = 60 * 24 * 7
    database_url: str = Field(..., validation_alias="DATABASE_URL")
    allowed_origins: list[str] | str = ["*"]
    gemini_api_key: str | None = Field(None, validation_alias="GEMINI_API_KEY")
    openai_api_key: str | None = Field(None, validation_alias="OPENAI_API_KEY")

    model_config = ConfigDict(env_file=".env", env_file_encoding="utf-8", extra="ignore")

    def get_allowed_origins(self) -> list[str]:
        raw_list = (
            [o.strip() for o in self.allowed_origins.split(",") if o.strip()]
            if isinstance(self.allowed_origins, str)
            else list(self.allowed_origins)
        )
        cleaned = set()
        for origin in raw_list:
            if origin == "*":
                cleaned.add("*")
            else:
                # Remove trailing slash because browsers send Origin without trailing slash
                cleaned.add(origin.rstrip("/"))
        return list(cleaned)


settings = Settings()

from datetime import datetime, timedelta, timezone

from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt

from app.core.config import settings
from app.models import User
from app.schemas.token import TokenPayload
from app.db.session import get_session
from sqlalchemy.orm import Session
from sqlalchemy import select

import bcrypt


def verify_password(plain_password: str, hashed_password: str) -> bool:
    try:
        return bcrypt.checkpw(
            plain_password.encode("utf-8")[:72],
            hashed_password.encode("utf-8"),
        )
    except Exception:
        return False


def get_password_hash(password: str) -> str:
    pwd_bytes = password.encode("utf-8")[:72]
    return bcrypt.hashpw(pwd_bytes, bcrypt.gensalt()).decode("utf-8")


oauth2_scheme = OAuth2PasswordBearer(tokenUrl=f"{settings.api_v1_str}/auth/login")
oauth2_optional_scheme = OAuth2PasswordBearer(tokenUrl=f"{settings.api_v1_str}/auth/login", auto_error=False)


def create_access_token(subject: str, expires_delta: timedelta | None = None) -> str:
    now = datetime.now(timezone.utc)
    expire = now + (expires_delta or timedelta(minutes=settings.access_token_expire_minutes))
    payload = {"sub": subject, "exp": int(expire.timestamp())}
    return jwt.encode(payload, settings.secret_key, algorithm="HS256")


def decode_access_token(token: str) -> TokenPayload:
    try:
        payload = jwt.decode(token, settings.secret_key, algorithms=["HS256"])
        sub = payload.get("sub")
        exp = payload.get("exp")
        if not sub or not exp:
            raise JWTError("Invalid token payload")
        exp_datetime = datetime.fromtimestamp(exp, tz=timezone.utc)
        token_data = TokenPayload(sub=str(sub), exp=exp_datetime)
    except JWTError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials",
            headers={"WWW-Authenticate": "Bearer"},
        ) from exc
    return token_data


def get_current_user(token: str = Depends(oauth2_scheme), session: Session = Depends(get_session)) -> User:
    token_data = decode_access_token(token)
    user = session.scalar(select(User).where(User.email == token_data.sub))
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return user


def get_optional_current_user(
    token: str | None = Depends(oauth2_optional_scheme),
    session: Session = Depends(get_session),
) -> User | None:
    if not token:
        return None
    try:
        token_data = decode_access_token(token)
        return session.scalar(select(User).where(User.email == token_data.sub))
    except Exception:
        return None

import re
import math


def slugify(value: str) -> str:
    text = value.strip().lower()
    text = re.sub(r"[^a-z0-9]+", "-", text)
    return re.sub(r"-+", "-", text).strip("-")


def calculate_word_count(body: str) -> int:
    return len(re.findall(r"\w+", body))


def calculate_read_time(body: str, words_per_minute: int = 200) -> int:
    words = calculate_word_count(body)
    return max(1, math.ceil(words / words_per_minute))

from sqlalchemy import create_engine
from sqlalchemy.orm import Session, sessionmaker

from app.core.config import settings

db_url = str(settings.database_url)
# Railway and Heroku often provide postgres:// or postgresql:// URLs
# Normalize to postgresql+psycopg:// if it's a PostgreSQL URL without an explicit driver
if db_url.startswith("postgres://"):
    db_url = db_url.replace("postgres://", "postgresql+psycopg://", 1)
elif db_url.startswith("postgresql://") and not db_url.startswith("postgresql+"):
    db_url = db_url.replace("postgresql://", "postgresql+psycopg://", 1)

engine_args = {"future": True}
if db_url.startswith("sqlite"):
    engine_args["connect_args"] = {"check_same_thread": False}

engine = create_engine(db_url, **engine_args)
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False, future=True, class_=Session)


def get_session():
    session = SessionLocal()
    try:
        yield session
    finally:
        session.close()

from datetime import datetime

import re
from pydantic import BaseModel, ConfigDict, EmailStr, Field, field_validator

# Strict Google Gmail username and domain compilation rules:
# 1. Domain must strictly be @gmail.com
# 2. Local part (username before @gmail.com) must be between 6 and 30 characters
# 3. Only alphanumeric characters (a-z, 0-9) and periods (.) are allowed (no underscores, hyphens, plus signs, etc.)
# 4. Cannot begin or end with a period (.)
# 5. Cannot have consecutive periods (..)
GMAIL_USERNAME_ALLOWED_CHARS = re.compile(r"^[a-z0-9.]+$")


def validate_gmail_address(value: str) -> str:
    cleaned = value.strip().lower()
    if not cleaned.endswith("@gmail.com"):
        raise ValueError("Fix email: only @gmail.com addresses are allowed (e.g. yourname@gmail.com)")

    username = cleaned[:-10]
    if len(username) < 6 or len(username) > 30:
        raise ValueError("Fix email: Gmail username must be between 6 and 30 characters long")

    if not GMAIL_USERNAME_ALLOWED_CHARS.match(username):
        raise ValueError("Fix email: Gmail username can only contain letters (a-z), numbers (0-9), and periods (.)")

    if username.startswith(".") or username.endswith("."):
        raise ValueError("Fix email: Gmail username cannot begin or end with a period")

    if ".." in username:
        raise ValueError("Fix email: Gmail username cannot contain consecutive periods (..)")

    return cleaned


class UserCreate(BaseModel):
    email: EmailStr
    username: str = Field(..., min_length=3, max_length=50)
    password: str = Field(..., min_length=8)

    @field_validator("email", mode="before")
    @classmethod
    def validate_email_format(cls, v: str) -> str:
        return validate_gmail_address(str(v))


class UserLogin(BaseModel):
    email: EmailStr
    password: str

    @field_validator("email", mode="before")
    @classmethod
    def validate_email_format(cls, v: str) -> str:

class UserUpdate(BaseModel):

    session: Session = Depends(get_session),
):
    if user_update.username is not None:
        new_username = user_update.username.strip()
        if not new_username:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Username cannot be empty",
            )
        # Check if username is already taken by another user
        if new_username != current_user.username:
            existing = get_user_by_username(session, new_username)
            if existing and existing.id != current_user.id:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Username is already taken",
                )
            current_user.username = new_username

    if user_update.bio is not None:
        current_user.bio = user_update.bio.strip() or None

    session.add(current_user)
    session.commit()
    session.refresh(current_user)
    return current_user

def _generate_with_gemini(api_key: str, prompt: str, system_instruction: str, response_schema: type) -> dict:
    """Invokes Google Gemini with candidate model fallback across 2.5, 2.0, and 1.5 flash."""
    from google import genai
    from google.genai import types

    client = genai.Client(api_key=api_key)
    candidate_models = ["gemini-2.5-flash", "gemini-2.0-flash", "gemini-1.5-flash"]
    last_exc = None

    for model_name in candidate_models:
        try:
            response = client.models.generate_content(
                model=model_name,
                contents=prompt,
                config=types.GenerateContentConfig(
                    system_instruction=system_instruction,
                    response_mime_type="application/json",
                    response_schema=response_schema,
                ),
            )
            return json.loads(response.text)
        except Exception as exc:
            last_exc = exc
            err_str = str(exc).lower()
            if "not found" in err_str or "404" in err_str or "unsupported" in err_str:
                logger.info("Model %s unavailable, trying next model: %s", model_name, exc)
                continue
            # If it's an authentication error or quota or bad request, re-raise immediately
            raise exc

    if last_exc:
        raise last_exc
