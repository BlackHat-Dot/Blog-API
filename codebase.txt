from pydantic import ConfigDict, Field, PostgresDsn
from pydantic_settings import BaseSettings


class Settings(BaseSettings):
    app_name: str = "Longhand API"
    api_v1_str: str = "/api/v1"
    secret_key: str = Field(..., validation_alias="SECRET_KEY")
    access_token_expire_minutes: int = 60 * 24 * 7
    database_url: str = Field(..., validation_alias="DATABASE_URL")
    allowed_origins: list[str] | str = ["*"]
    gemini_api_key: str | None = Field(None, validation_alias="GEMINI_API_KEY")
    openai_api_key: str | None = Field(None, validation_alias="OPENAI_API_KEY")

    model_config = ConfigDict(env_file=".env", env_file_encoding="utf-8", extra="ignore")

    def get_allowed_origins(self) -> list[str]:
        raw_list = (
            [o.strip() for o in self.allowed_origins.split(",") if o.strip()]
            if isinstance(self.allowed_origins, str)
            else list(self.allowed_origins)
        )
        cleaned = set()
        for origin in raw_list:
            if origin == "*":
                cleaned.add("*")
            else:
                # Remove trailing slash because browsers send Origin without trailing slash
                cleaned.add(origin.rstrip("/"))
        return list(cleaned)


settings = Settings()

from datetime import datetime, timedelta, timezone

from fastapi import Depends, HTTPException, status

from sqlalchemy.orm import Session
from sqlalchemy import select

import bcrypt


def verify_password(plain_password: str, hashed_password: str) -> bool:
    try:
        return bcrypt.checkpw(
            plain_password.encode("utf-8")[:72],
            hashed_password.encode("utf-8"),
        )
    except Exception:
        return False


def get_password_hash(password: str) -> str:
    pwd_bytes = password.encode("utf-8")[:72]
    return bcrypt.hashpw(pwd_bytes, bcrypt.gensalt()).decode("utf-8")


oauth2_scheme = OAuth2PasswordBearer(tokenUrl=f"{settings.api_v1_str}/auth/login")
oauth2_optional_scheme = OAuth2PasswordBearer(tokenUrl=f"{settings.api_v1_str}/auth/login", auto_error=False)


def create_access_token(subject: str, expires_delta: timedelta | None = None) -> str:
    now = datetime.now(timezone.utc)
    expire = now + (expires_delta or timedelta(minutes=settings.access_token_expire_minutes))
    payload = {"sub": subject, "exp": int(expire.timestamp())}
    return jwt.encode(payload, settings.secret_key, algorithm="HS256")


def decode_access_token(token: str) -> TokenPayload:
    try:
        payload = jwt.decode(token, settings.secret_key, algorithms=["HS256"])
        sub = payload.get("sub")
        exp = payload.get("exp")
        if not sub or not exp:
            raise JWTError("Invalid token payload")
        exp_datetime = datetime.fromtimestamp(exp, tz=timezone.utc)
        token_data = TokenPayload(sub=str(sub), exp=exp_datetime)
    except JWTError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials",
            headers={"WWW-Authenticate": "Bearer"},
        ) from exc
    return token_data


def get_current_user(token: str = Depends(oauth2_scheme), session: Session = Depends(get_session)) -> User:
    token_data = decode_access_token(token)
    user = session.scalar(select(User).where(User.email == token_data.sub))
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return user


def get_optional_current_user(
    token: str | None = Depends(oauth2_optional_scheme),
    session: Session = Depends(get_session),
) -> User | None:
    if not token:
        return None
    try:
        token_data = decode_access_token(token)
        return session.scalar(select(User).where(User.email == token_data.sub))
    except Exception:
        return None

import re
import math


def slugify(value: str) -> str:
    text = value.strip().lower()
    text = re.sub(r"[^a-z0-9]+", "-", text)
    return re.sub(r"-+", "-", text).strip("-")


def calculate_word_count(body: str) -> int:
    return len(re.findall(r"\w+", body))


def calculate_read_time(body: str, words_per_minute: int = 200) -> int:
    words = calculate_word_count(body)
    return max(1, math.ceil(words / words_per_minute))

from sqlalchemy import create_engine
from sqlalchemy.orm import Session, sessionmaker

from app.core.config import settings

db_url = str(settings.database_url)
# Railway and Heroku often provide postgres:// or postgresql:// URLs
# Normalize to postgresql+psycopg:// if it's a PostgreSQL URL without an explicit driver
if db_url.startswith("postgres://"):
    db_url = db_url.replace("postgres://", "postgresql+psycopg://", 1)
elif db_url.startswith("postgresql://") and not db_url.startswith("postgresql+"):
    db_url = db_url.replace("postgresql://", "postgresql+psycopg://", 1)

engine_args = {"future": True}
if db_url.startswith("sqlite"):
    engine_args["connect_args"] = {"check_same_thread": False}

engine = create_engine(db_url, **engine_args)
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False, future=True, class_=Session)


def get_session():
    session = SessionLocal()
    try:
        yield session
    finally:
        session.close()

from datetime import datetime

import re
from pydantic import BaseModel, ConfigDict, EmailStr, Field, field_validator

# Strict Google Gmail username and domain compilation rules:
# 1. Domain must strictly be @gmail.com
# 2. Local part (username before @gmail.com) must be between 6 and 30 characters
# 3. Only alphanumeric characters (a-z, 0-9) and periods (.) are allowed (no underscores, hyphens, plus signs, etc.)
# 4. Cannot begin or end with a period (.)
# 5. Cannot have consecutive periods (..)
GMAIL_USERNAME_ALLOWED_CHARS = re.compile(r"^[a-z0-9.]+$")


def validate_gmail_address(value: str) -> str:
    cleaned = value.strip().lower()
    if not cleaned.endswith("@gmail.com"):
        raise ValueError("Fix email: only @gmail.com addresses are allowed (e.g. yourname@gmail.com)")

    username = cleaned[:-10]
    if len(username) < 6 or len(username) > 30:
        raise ValueError("Fix email: Gmail username must be between 6 and 30 characters long")

    if not GMAIL_USERNAME_ALLOWED_CHARS.match(username):
        raise ValueError("Fix email: Gmail username can only contain letters (a-z), numbers (0-9), and periods (.)")

    if username.startswith(".") or username.endswith("."):
        raise ValueError("Fix email: Gmail username cannot begin or end with a period")

    if ".." in username:
        raise ValueError("Fix email: Gmail username cannot contain consecutive periods (..)")

    return cleaned


class UserCreate(BaseModel):
    email: EmailStr
    username: str = Field(..., min_length=3, max_length=50)
    password: str = Field(..., min_length=8)

    @field_validator("email", mode="before")
    @classmethod
    def validate_email_format(cls, v: str) -> str:
        return validate_gmail_address(str(v))


class UserLogin(BaseModel):
    email: EmailStr
    password: str

    @classmethod

def _normalize_tag_name(value: str) -> str:
    return value.strip().lower()


def _generate_unique_slug(session: Session, title: str, exclude_post_id: int | None = None) -> str:
    base_slug = slugify(title)
    slug = base_slug
    index = 1
    while True:
        existing = session.scalar(select(Post).where(Post.slug == slug))
        if existing is None or (exclude_post_id and existing.id == exclude_post_id):
            break
        index += 1
        slug = f"{base_slug}-{index}"
    return slug


def get_category(session: Session, category_id: int) -> Category | None:
    return session.get(Category, category_id)


def get_or_create_tags(session: Session, names: list[str]) -> list[Tag]:
    tags: list[Tag] = []
    for name in {name.strip().lower() for name in names if name.strip()}:
        existing = session.scalar(select(Tag).where(Tag.name == name))
        if existing:
            tags.append(existing)
            continue
        slug = slugify(name)
        if session.scalar(select(Tag).where(Tag.slug == slug)):
            slug = f"{slug}-{len(name)}"
        tag = Tag(name=name, slug=slug)
        session.add(tag)
        session.flush()
        tags.append(tag)
    return tags


def create_post(session: Session, author_id: int, post_data: PostCreate) -> Post:
    slug = _generate_unique_slug(session, post_data.title)
    word_count = calculate_word_count(post_data.body)
    read_time = calculate_read_time(post_data.body)
    post = Post(
        title=post_data.title,
        summary=post_data.summary,
        body=post_data.body,
        slug=slug,
        state=post_data.state,
        word_count=word_count,
        read_time=read_time,
        author_id=author_id,
        category_id=post_data.category_id,
    )
    post.tags = get_or_create_tags(session, post_data.tags)
    session.add(post)
    session.commit()
    return get_post(session, post.id)


def update_post(session: Session, post: Post, post_data: PostUpdate) -> Post:
    if post_data.title:
        post.title = post_data.title
        post.slug = _generate_unique_slug(session, post_data.title, exclude_post_id=post.id)
    if post_data.summary is not None:
        post.summary = post_data.summary
    if post_data.body is not None:
        post.body = post_data.body
        post.word_count = calculate_word_count(post.body)
        post.read_time = calculate_read_time(post.body)
    if post_data.state is not None:
        post.state = post_data.state
    if post_data.category_id is not None:
        post.category_id = post_data.category_id
    if post_data.tags is not None:
        post.tags = get_or_create_tags(session, post_data.tags)
    session.add(post)
    session.commit()
    return get_post(session, post.id)

def list_posts(
    session: Session,
    search: str | None = None,
    category: str | None = None,
    tag: str | None = None,
    author: str | None = None,
    state: str = "published",
    limit: int = 20,
    offset: int = 0,
):
    statement = (
        select(Post)
        .options(
            selectinload(Post.author),
            selectinload(Post.category),
            selectinload(Post.tags),
            selectinload(Post.likes),
            selectinload(Post.bookmarks),
            selectinload(Post.comments),
        )
        .order_by(Post.created_at.desc())
    )
    count_query = select(func.count(func.distinct(Post.id))).select_from(Post)

    if search:
        search_filter = (
            Post.title.ilike(f"%{search}%")
            | Post.summary.ilike(f"%{search}%")
            | Post.body.ilike(f"%{search}%")
        )
        statement = statement.where(search_filter)
        count_query = count_query.where(search_filter)

    if state:
        statement = statement.where(Post.state == state)
        count_query = count_query.where(Post.state == state)

    if category:
        statement = statement.join(Post.category).where(Category.slug == category)
        count_query = count_query.join(Post.category).where(Category.slug == category)

    if tag:
        statement = statement.join(Post.tags).where(Tag.slug == tag)
        count_query = count_query.join(Post.tags).where(Tag.slug == tag)

    if author:
        statement = statement.join(Post.author).where(User.username == author)
        count_query = count_query.join(Post.author).where(User.username == author)

    if category or tag or author:
        statement = statement.distinct()

    total = session.scalar(count_query) or 0
    posts = session.scalars(statement.limit(limit).offset(offset)).all()
    return posts, total


def get_popular_posts(session: Session, limit: int = 10, offset: int = 0):
    total = session.scalar(select(func.count(Post.id)).where(Post.state == "published")) or 0
    statement = (
        select(Post, func.count(Like.id).label("like_count"))
        .options(
            selectinload(Post.author),
            selectinload(Post.category),
            selectinload(Post.tags),
            selectinload(Post.likes),
            selectinload(Post.bookmarks),
            selectinload(Post.comments),
        )
        .outerjoin(Like, Like.post_id == Post.id)
        .where(Post.state == "published")
        .group_by(Post.id)
        .order_by(func.count(Like.id).desc(), Post.created_at.desc())
        .limit(limit)
        .offset(offset)
    )
    rows = session.execute(statement).all()
    return rows, total

def create_comment(session: Session, post_id: int, author_id: int, comment_data: CommentCreate) -> Comment:
    if not session.get(Post, post_id):
        raise ValueError("Post not found")
    if comment_data.parent_id is not None:
        parent_comment = session.get(Comment, comment_data.parent_id)
        if not parent_comment:
            raise ValueError("Parent comment not found")
        if parent_comment.post_id != post_id:
            raise ValueError("Parent comment belongs to a different post")
    comment = Comment(
        body=comment_data.body,
        post_id=post_id,
        author_id=author_id,
        parent_id=comment_data.parent_id,
    )
    session.add(comment)
    session.commit()
    statement = select(Comment).options(selectinload(Comment.author)).where(Comment.id == comment.id)
    return session.scalar(statement)


def list_comments(session: Session, post_id: int, limit: int = 20, offset: int = 0) -> list[Comment]:
    statement = (
        select(Comment)
        .options(selectinload(Comment.author))
        .where(Comment.post_id == post_id)
        .order_by(Comment.created_at.asc())
        .limit(limit)
        .offset(offset)
    )
    return session.scalars(statement).all()

def get_follow(session: Session, follower_id: int, followed_id: int) -> Follow | None:
    return session.scalar(
        select(Follow).where(Follow.follower_id == follower_id, Follow.followed_id == followed_id)
    )


def follow_author(session: Session, follower_id: int, followed_id: int) -> Follow:
    if follower_id == followed_id:
        raise ValueError("You cannot follow yourself")
    follow = get_follow(session, follower_id, followed_id)
    if follow:
        return follow
    follow = Follow(follower_id=follower_id, followed_id=followed_id)
    session.add(follow)
    session.commit()
    session.refresh(follow)
    return follow


def unfollow_author(session: Session, follower_id: int, followed_id: int) -> bool:
    follow = get_follow(session, follower_id, followed_id)
    if not follow:
        return False
    session.delete(follow)
    session.commit()
    return True

@router.get("/users/me", response_model=UserRead)
def read_current_user(current_user=Depends(get_current_user)):
    return current_user


@router.put("/users/me", response_model=UserRead)
def update_current_user(
    user_update: UserUpdate,
    current_user: User = Depends(get_current_user),
    session: Session = Depends(get_session),
):
    if user_update.username is not None:
        new_username = user_update.username.strip()
        if not new_username:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Username cannot be empty",
            )
        # Check if username is already taken by another user
        if new_username != current_user.username:
            existing = get_user_by_username(session, new_username)
            if existing and existing.id != current_user.id:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Username is already taken",
                )
            current_user.username = new_username

    if user_update.bio is not None:
        current_user.bio = user_update.bio.strip() or None

    session.add(current_user)
    session.commit()
    session.refresh(current_user)
    return current_user


@router.get("/profiles/{username}", response_model=Profile)
def read_profile(
    username: str,
    session: Session = Depends(get_session),
    viewer: User | None = Depends(get_optional_current_user),
):
    user = get_user_by_username(session, username)
    if not user:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Author not found")

    post_count = session.scalar(select(func.count()).select_from(Post).where(Post.author_id == user.id)) or 0
    follower_count = session.scalar(select(func.count()).select_from(Follow).where(Follow.followed_id == user.id)) or 0
    following_count = session.scalar(select(func.count()).select_from(Follow).where(Follow.follower_id == user.id)) or 0

    is_following = False
    if viewer and viewer.id != user.id:
        is_following = bool(get_follow(session, viewer.id, user.id))

    return Profile(
        username=user.username,
        bio=user.bio,
        post_count=post_count,
        follower_count=follower_count,
        following_count=following_count,
        is_following=is_following,
        created_at=user.created_at,
    )


@router.post("/profiles/{username}/follow", status_code=status.HTTP_204_NO_CONTENT)
def follow(username: str, current_user=Depends(get_current_user), session: Session = Depends(get_session)):
    user = get_user_by_username(session, username)
    if not user:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Author not found")
    if user.id == current_user.id:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail="Cannot follow yourself")
    follow_author(session, current_user.id, user.id)
    return None


@router.delete("/profiles/{username}/follow", status_code=status.HTTP_204_NO_CONTENT)
def unfollow(username: str, current_user=Depends(get_current_user), session: Session = Depends(get_session)):
    user = get_user_by_username(session, username)
    if not user:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Author not found")
    if not unfollow_author(session, current_user.id, user.id):
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail="Not following this author")
    return None

def _build_post_response(
    post,
    like_count: int | None = None,
    bookmark_count: int | None = None,
    viewer: User | None = None,
    is_liked: bool | None = None,
    is_bookmarked: bool | None = None,
) -> PostResponse:
    if is_liked is None:
        is_liked = any(l.user_id == viewer.id for l in post.likes) if viewer else False
    if is_bookmarked is None:
        is_bookmarked = any(b.user_id == viewer.id for b in post.bookmarks) if viewer else False

    return PostResponse(
        id=post.id,
        title=post.title,
        summary=post.summary,
        body=post.body,
        slug=post.slug,
        state=post.state,
        word_count=post.word_count,
        read_time=post.read_time,
        author={"username": post.author.username, "bio": post.author.bio},
        category=CategoryRead.model_validate(post.category) if post.category else None,
        tags=[tag.name for tag in post.tags],
        like_count=like_count if like_count is not None else len(post.likes),
        bookmark_count=bookmark_count if bookmark_count is not None else len(post.bookmarks),
        comment_count=len(post.comments),
        created_at=post.created_at,
        updated_at=post.updated_at,
        is_liked=is_liked,
        is_bookmarked=is_bookmarked,
    )

    session.add(category)
    session.commit()

def _generate_post_fallback(prompt: str, tone: str, existing_categories: list[str]) -> GeneratedPostResponse:
    """High quality structured fallback generator when no external API key is active."""
    clean_prompt = prompt.strip().rstrip(".")
    title_words = [w.capitalize() for w in clean_prompt.split()]
    title = f"{' '.join(title_words[:8])}: A Study in {tone.split()[0]}"
    if len(title) < 15:
        title = f"The Art of {clean_prompt.capitalize()}"

    summary = (
        f"A thorough, impeccably written exploration of {clean_prompt.lower()}, "
        f"approached through the distinctive lens of a {tone.lower()}."
    )

    category = "Technology"
    if existing_categories:
        for cat in existing_categories:
            if cat.lower() in prompt.lower() or cat.lower() in tone.lower():
                category = cat
                break
        else:
            category = existing_categories[0]

    # Generate rich markdown body
    body = f"""## The Genesis of the Idea

Every compelling narrative begins with a fundamental tension. When examining **{clean_prompt}**, the prevailing consensus often overlooks the subtle nuances that govern how systems and minds actually operate in practice. 

Approaching this from a **{tone}** perspective demands that we strip away performative rhetoric and focus squarely on firsthand substance.

> "True clarity is not the absence of complexity, but the mastery of its constituent elements."

## Deconstructing the Underlying Principles

To genuinely master this domain, consider the three core pillars that underpin the argument:

1. **Precision of Intent**: Moving beyond superficial answers toward principled foundations.
2. **Contextual Execution**: Realizing that dogma frequently fails when applied blindly across heterogeneous environments.
3. **Iterative Refinement**: Embracing continuous measurement, candid reflection, and disciplined iteration.

Whether you are building resilient architectures or crafting strategic narratives, the friction is rarely in the destination—it is in the unwarranted assumptions carried along the journey.

---

## Practical Takeaways and Next Horizons

As we synthesize these observations, the overarching lesson remains clear: rigor in thought breeds effortless execution. 

- Re-examine your baseline constraints without deference to legacy habits.
- Test assumptions under stress before scaling them across production.
- Document insights clearly so the rationale endures long after the initial momentum subsides.
"""

    tags = [re.sub(r"[^a-z0-9]", "", w.lower()) for w in clean_prompt.split() if len(w) > 3][:4]
    if not tags:
        tags = ["essay", "longhand", "perspective"]

    return GeneratedPostResponse(
        title=title,
        summary=summary,
        body=body.strip(),
        category=category,
        tags=tags,
    )

def test_ai_tones_and_post_generation(client):
    # Setup user
    client.post(
        "/api/v1/auth/register",
        json={"email": "aiwriter@gmail.com", "username": "aiwriter", "password": "Password123!"},
    )
    login_res = client.post(
        "/api/v1/auth/login",
        json={"email": "aiwriter@gmail.com", "password": "Password123!"},
    )
    headers = {"Authorization": f"Bearer {login_res.json()['access_token']}"}

    # Test suggest-tones without auth (401)
    unauth_tones = client.post("/api/v1/ai/suggest-tones", json={"prompt": "Why SQLite in production is awesome"})
    assert unauth_tones.status_code == 401

    # Test suggest-tones with auth (200)
    tones_res = client.post(
        "/api/v1/ai/suggest-tones",
        json={"prompt": "Why SQLite in production is awesome"},
        headers=headers,
    )
    assert tones_res.status_code == 200
    tones_data = tones_res.json()
    assert "tones" in tones_data
    assert len(tones_data["tones"]) >= 3
    selected_tone = tones_data["tones"][0]["name"]
    assert len(selected_tone) > 0
    assert "description" in tones_data["tones"][0]

    # Test generate-post without auth (401)
    unauth_gen = client.post(
        "/api/v1/ai/generate-post",
        json={"prompt": "Why SQLite in production is awesome", "tone": selected_tone},
    )
    assert unauth_gen.status_code == 401

    # Test generate-post with auth (200)
    gen_res = client.post(
        "/api/v1/ai/generate-post",
        json={
            "prompt": "Why SQLite in production is awesome",
            "tone": selected_tone,
            "existing_categories": ["Technology", "Architecture"],
        },
        headers=headers,
    )
    assert gen_res.status_code == 200
    gen_data = gen_res.json()
    assert "title" in gen_data and len(gen_data["title"]) >= 5
    assert "summary" in gen_data and len(gen_data["summary"]) >= 10
    assert "body" in gen_data and len(gen_data["body"]) >= 50
    assert "##" in gen_data["body"]  # Verify formatted markdown
    assert isinstance(gen_data["tags"], list)
    assert len(gen_data["tags"]) >= 1
    assert gen_data["category"] in ["Technology", "Architecture"]
